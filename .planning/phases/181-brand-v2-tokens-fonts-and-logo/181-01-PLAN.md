MailglassAdmin.TestSupport.Contrast
  normalize_hex(String.t()) :: String.t()            # "#fff" / "#FFFFFF" -> "#ffffff"
  luminance(hex) :: float                            # WCAG 2.2 relative luminance, sRGB threshold 0.04045
  ratio(fg_hex, bg_hex) :: float                     # (L1 + 0.05) / (L2 + 0.05), order-independent
  composite(fg_hex, bg_hex, alpha :: float) :: hex   # per channel round(alpha*fg + (1-alpha)*bg), Elixir round/1
  resolve_tokens(tokens_json_map) :: %{{tier, css_var} => hex}
      # tier in "light" | "dark"; css_var "--mg-color-<role>" from color.<tier>.<role>,
      # "--mg-logo-<role>" from logo.<tier>.<role>; resolves "{palette.x}" aliases; raises on a dangling alias
